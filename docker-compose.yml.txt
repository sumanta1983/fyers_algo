services:
  postgres:
    image: postgres:16-alpine
    container_name: fyers_algo_postgres
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-postgres}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD:-postgres}
      POSTGRES_DB: ${POSTGRES_DB:-fyers_algo}
      TZ: ${TZ:-Asia/Kolkata}
    ports:
      - "${POSTGRES_PORT:-5432}:5432"
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./node_backend/migrations:/docker-entrypoint-initdb.d:ro
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-postgres} -d ${POSTGRES_DB:-fyers_algo}"]
      interval: 5s
      timeout: 3s
      retries: 10

  redis:
    image: redis:7-alpine
    container_name: fyers_algo_redis
    restart: unless-stopped
    command: redis-server --appendonly no --save "" --requirepass "${REDIS_PASSWORD:-}"
    environment:
      # redis-cli inside the container reads REDISCLI_AUTH automatically,
      # so the healthcheck works whether or not a password is set.
      REDISCLI_AUTH: ${REDIS_PASSWORD:-}
    ports:
      - "${REDIS_PORT:-6379}:6379"
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 5s
      timeout: 3s
      retries: 10

  postgres_data:
